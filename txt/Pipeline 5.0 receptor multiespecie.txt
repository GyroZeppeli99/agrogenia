watch -n 15 'ls -lh /tmp/tmp8zupby7d.fasta.domtbl 2>/dev/null || echo "domtbl aún no aparece"'

watch -n 15 'ps -p 203368 -o pid,etime,time,%cpu,comm; echo; ls -lh /tmp/tmp8zupby7d.fasta.domtbl 2>/dev/null || echo "domtbl aún no aparece"'
